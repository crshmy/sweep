def load_json_any(fp: str):
    """UTF-8 → 실패 시 UTF-8-SIG(BOM)로 재시도"""
    if not os.path.isfile(fp):
        raise FileNotFoundError(fp)
    try:
        with open(fp, "r", encoding="utf-8") as f:
            return json.load(f)
    except json.JSONDecodeError:
        with open(fp, "r", encoding="utf-8-sig") as f:
            return json.load(f)

def haversine_m(lon1, lat1, lon2, lat2) -> float:
    R = 6371000.0
    dlon = math.radians(lon2 - lon1)
    dlat = math.radians(lat2 - lat1)
    a = math.sin(dlat/2.0)**2 + math.cos(math.radians(lat1))*math.cos(math.radians(lat2))*math.sin(dlon/2.0)**2
    c = 2.0 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
    return R * c

def bbox_expand(lons, lats, pad_deg=0.1):
    return (min(lons)-pad_deg, min(lats)-pad_deg, max(lons)+pad_deg, max(lats)+pad_deg)

def random_point_in_bbox(bbox):
    minx, miny, maxx, maxy = bbox
    return (random.uniform(minx, maxx), random.uniform(miny, maxy))

def move_by_gaussian(lon, lat, sigma_m=200.0):
    dlat = (sigma_m / 111000.0) * np.random.normal()
    dlon = (sigma_m / (111000.0 * max(1e-6, math.cos(math.radians(lat))))) * np.random.normal()
    return lon + dlon, lat + dlat

def chaikin(points: List[Tuple[float,float]], t=0.25, iterations=1):
    if len(points) < 3 or iterations <= 0:
        return points[:]
    pts = points[:]
    for _ in range(iterations):
        new_pts = [pts[0]]
        for i in range(len(pts)-1):
            p0 = np.array(pts[i])
            p1 = np.array(pts[i+1])
            Q = (1-t)*p0 + t*p1
            R = t*p0 + (1-t)*p1
            new_pts.extend([tuple(Q), tuple(R)])
        new_pts.append(pts[-1])
        pts = new_pts
    return pts

def load_hotspots(fp: str) -> List[Hotspot]:
    raw = load_json_any(fp)
    hs_list: List[Hotspot] = []
    hs_raw = raw.get("hotspots", [])
    for i, rec in enumerate(hs_raw):
        lat = rec.get("lat")
        lon = rec.get("lng", rec.get("lon"))
        if lat is None or lon is None:
            print(f"[hotspots] WARN: hotspot #{i} missing lat/lon → skip")
            continue
        try:
            lat = float(lat); lon = float(lon)
        except Exception:
            print(f"[hotspots] WARN: hotspot #{i} lat/lon not float → skip")
            continue
        w = rec.get("weight", 1.0)
        try:
            w = float(w)
        except Exception:
            w = 1.0
        mass = max(1.0, 5.0 * w)
        hs_list.append(Hotspot(id=f"hs_{i}", lon=lon, lat=lat, weight=w, mass=mass))

    if not hs_list:
        raise ValueError("[hotspots] No valid hotspot records.")

    print(f"[hotspots] OK: loaded {len(hs_list)} records from {fp}")
    return hs_list

def load_risk_points(fp: str) -> List[Tuple[float,float,float]]:
    try:
        gj = load_json_any(fp)
    except FileNotFoundError:
        print(f"[risk] WARN: file not found: {fp} → risk disabled")
        return []
    feats = gj.get("features", [])
    out = []
    for idx, ft in enumerate(feats):
        geom = ft.get("geometry", {})
        if geom.get("type") != "Point":
            continue
        coords = geom.get("coordinates", [])
        if len(coords) != 2:
            continue
        lon, lat = coords
        try:
            lon = float(lon); lat = float(lat)
            if math.isnan(lon) or math.isnan(lat):
                continue
        except Exception:
            continue
        prop = ft.get("properties", {})
        risk = prop.get("최종위험도", prop.get("risk", 0.0))
        try:
            risk = float(risk)
        except Exception:
            risk = 0.0
        out.append((lon, lat, risk))
    print(f"[risk] OK: loaded {len(out)} risk points from {fp}")
    return out

class RiskField:
    def __init__(self, pts: List[Tuple[float,float,float]]):
        self.pts = np.array(pts, dtype=float) if pts else np.zeros((0,3), dtype=float)
    def risk_at(self, lon: float, lat: float, neigh_r_m: float = RISK_NEIGHBOR_R_M) -> float:
        if self.pts.size == 0:
            return 0.0
        dlat_deg = neigh_r_m / 111000.0
        dlon_deg = neigh_r_m / (111000.0 * max(1e-6, math.cos(math.radians(lat))))
        mask = (
            (self.pts[:,0] >= lon - dlon_deg) &
            (self.pts[:,0] <= lon + dlon_deg) &
            (self.pts[:,1] >= lat - dlat_deg) &
            (self.pts[:,1] <= lat + dlat_deg)
        )
        cand = self.pts[mask]
        if cand.size == 0:
            return 0.0
        dists = np.array([haversine_m(lon, lat, L, A) for L, A, _ in cand], dtype=float)
        w = np.exp(-np.clip(dists, 0, neigh_r_m) / max(1.0, 0.35*neigh_r_m))
        risks = cand[:,2]
        denom = w.sum()
        return float((w * risks).sum() / denom) if denom > 0 else float(risks.mean())

# 경로 스코어/선택
# =========================
def score_hotspot(curr: Tuple[float,float], hs: Hotspot, risk_field: RiskField) -> float:
    if hs.visited:
        return -1e9
    dist = haversine_m(curr[0], curr[1], hs.lon, hs.lat) + 1.0
    local_risk = risk_field.risk_at(hs.lon, hs.lat)
    importance = 0.4 * hs.weight + 0.6 * hs.mass
    cost = dist * (1.0 + RISK_WEIGHT * local_risk)
    return importance / cost

def score_random_trash(curr: Tuple[float,float], trash: TrashPoint) -> float:
    if trash.visited:
        return -1e9
    dist = haversine_m(curr[0], curr[1], trash.lon, trash.lat) + 1.0
    return trash.weight / dist

def select_next_target(curr_lon: float, curr_lat: float,
                       hotspots: List[Hotspot], 
                       rand_pts: List[TrashPoint],
                       risk_field: RiskField) -> Optional[Tuple[str, object]]:
    best_type = None
    best_obj = None
    best_score = -1e18
    
    for h in hotspots:
        if h.visited:
            continue
        s = score_hotspot((curr_lon, curr_lat), h, risk_field)
        if s > best_score:
            best_score = s
            best_type = 'hotspot'
            best_obj = h
    
    for rp in rand_pts:
        if rp.visited:
            continue
        s = score_random_trash((curr_lon, curr_lat), rp)
        if s > best_score:
            best_score = s
            best_type = 'random'
            best_obj = rp
    
    if best_obj is None:
        return None
    return (best_type, best_obj)

def generate_random_trash(hotspots: List[Hotspot],
                          bbox: Optional[Tuple[float,float,float,float]] = None) -> List[TrashPoint]:
    if not hotspots:
        return []
    hs_lons = [h.lon for h in hotspots]
    hs_lats = [h.lat for h in hotspots]
    if bbox is None:
        bbox = bbox_expand(hs_lons, hs_lats, pad_deg=0.1)

    pts: List[TrashPoint] = []
    # 핫스팟 중심 가우시안
    for h in hotspots:
        for _ in range(int(RANDOM_GAUSS_PER_HS * (1.0 - RANDOM_UNIFORM_FRACTION))):
            lon, lat = move_by_gaussian(h.lon, h.lat, GAUSS_SIGMA_M)
            pts.append(TrashPoint(lon=lon, lat=lat, weight=1.0, is_cover=False))
    # 균일 일부
    for _ in range(int(RANDOM_UNIFORM_COUNT * RANDOM_UNIFORM_FRACTION)):
        lon, lat = random_point_in_bbox(bbox)
        pts.append(TrashPoint(lon=lon, lat=lat, weight=1.0, is_cover=False))
    return pts

def plan_route() -> dict:
    """
    스코어 기반 경로 계획
    - 거리, 위험도, 중요도를 종합 평가
    - 시간 제약 없이 모든 쓰레기 수거
    - 시작점: 전체(핫스팟+랜덤) 중 위도 최대 포인트
    """
    # 1. 데이터 로드
    print("\n=== LOADING DATA ===")
    hotspots = load_hotspots(HOTSPOT_JSON)
    risk_pts = load_risk_points(RISK_GEOJSON)
    risk_field = RiskField(risk_pts)
    
    # 2. 랜덤 쓰레기 생성
    print("\n=== GENERATING RANDOM TRASH ===")
    rand_pts = generate_random_trash(hotspots)
    print(f"[trash] Generated {len(rand_pts)} random trash points")
    
    # 3. 랜덤 쓰레기 → 최근접 핫스팟에 질량 귀속
    print("\n=== ASSIGNING RANDOM TRASH TO HOTSPOTS ===")
    non_cover_random = [t for t in rand_pts if not t.is_cover]
    hs_xy = np.array([(h.lon, h.lat) for h in hotspots], dtype=float)
    for t in non_cover_random:
        d = np.hypot((hs_xy[:,0]-t.lon)*math.cos(math.radians(t.lat))*111000.0,
                     (hs_xy[:,1]-t.lat)*111000.0)
        j = int(np.argmin(d))
        hotspots[j].mass += max(1.0, t.weight * 1.0)
    
    # 4. 출발점 결정: 전체(핫스팟 + 랜덤) 중 '위도' 최댓값
    print("\n=== DETERMINING START POINT (NORTHERNMOST TRASH) ===")
    candidates = [('hotspot', h) for h in hotspots] + [('random', r) for r in non_cover_random]
    if not candidates:
        raise ValueError("No hotspots or random trash points to start from.")
    start_type, start_obj = max(candidates, key=lambda x: x[1].lat)
    print(f"[start] Northernmost {start_type}: lat={start_obj.lat:.5f}, lon={start_obj.lon:.5f}")

    # 5. 경로 계획 (스코어 기반 선택)
    print("\n=== PLANNING ROUTE (SCORE-BASED) ===")
    route = [(start_obj.lon, start_obj.lat)]
    cur_lon, cur_lat = start_obj.lon, start_obj.lat
    
    total_travel_time = 0.0
    total_collection_time = 0.0
    total_distance_m = 0.0
    visited_count = 1
    collision_count = 0

    # 시작점 처리
    if start_type == 'hotspot':
        start_obj.visited = True
        start_obj.collected_mass = start_obj.mass
        # 반경 내 랜덤 쓰레기 일괄 수거
        capture_radius = max(CAPTURE_MIN_R_M, start_obj.radius_m * CAPTURE_FACTOR)
        rand_collected = 0
        for rp in non_cover_random:
            if not rp.visited:
                d = haversine_m(cur_lon, cur_lat, rp.lon, rp.lat)
                if d <= capture_radius:
                    rp.visited = True
                    rand_collected += 1
                    total_collection_time += COLLECTION_TIME_PER_RANDOM_S
                    start_obj.collected_mass += max(1.0, rp.weight * 1.0)
        # 핫스팟 자체 수거 시간
        total_collection_time += COLLECTION_TIME_PER_HOTSPOT_S
        print(f"[route] Visit #1: {start_obj.id} (START hotspot) | mass={start_obj.mass:.1f} | bonus_rand={rand_collected}")
    else:
        # random 시작
        start_obj.visited = True
        total_collection_time += COLLECTION_TIME_PER_RANDOM_S
        print(f"[route] Visit #1: random (START)")

    # 6. 나머지 모든 쓰레기 방문 (스코어 기반)
    iter_count = 0
    while True:
        next_target = select_next_target(cur_lon, cur_lat, hotspots, non_cover_random, risk_field)
        if next_target is None:
            print("[route] All trash collected!")
            break
        
        target_type, target_obj = next_target
        target_lon, target_lat = target_obj.lon, target_obj.lat
        
        # 이동
        dist = haversine_m(cur_lon, cur_lat, target_lon, target_lat)
        travel_time = dist / SPEED_MS
        collection_time = COLLECTION_TIME_PER_HOTSPOT_S if target_type == 'hotspot' else COLLECTION_TIME_PER_RANDOM_S
        
        # 방문
        target_obj.visited = True
        route.append((target_lon, target_lat))
        cur_lon, cur_lat = target_lon, target_lat
        total_distance_m += dist
        total_travel_time += travel_time
        total_collection_time += collection_time
        visited_count += 1
        
        if target_type == 'hotspot':
            # 핫스팟 수거
            target_obj.collected_mass = target_obj.mass
            # 반경 내 랜덤 쓰레기 일괄 수거
            capture_radius = max(CAPTURE_MIN_R_M, target_obj.radius_m * CAPTURE_FACTOR)
            rand_collected = 0
            for rp in non_cover_random:
                if not rp.visited:
                    d = haversine_m(cur_lon, cur_lat, rp.lon, rp.lat)
                    if d <= capture_radius:
                        rp.visited = True
                        rand_collected += 1
                        total_collection_time += COLLECTION_TIME_PER_RANDOM_S
                        target_obj.collected_mass += max(1.0, rp.weight * 1.0)
            # 위험도 체크
            risk_val = risk_field.risk_at(target_lon, target_lat)
            if risk_val > 0.5:
                collision_count += 1
            
            total_time_h = (total_travel_time + total_collection_time) / 3600.0
            print(f"[route] Visit #{visited_count}: {target_obj.id} (hotspot) | dist={dist:.0f}m | "
                  f"mass={target_obj.mass:.1f} | bonus_rand={rand_collected} | "
                  f"risk={risk_val:.2f} | time={total_time_h:.2f}h")
        else:
            total_time_h = (total_travel_time + total_collection_time) / 3600.0
            if visited_count % 50 == 0:
                print(f"[route] Visit #{visited_count}: random trash | dist={dist:.0f}m | time={total_time_h:.2f}h")
        
        iter_count += 1
        if iter_count > 10000:
            print("[route] WARNING: Max iterations reached")
            break
    
    # 7. 경로 스무딩
    print("\n=== SMOOTHING ROUTE ===")
    route_smooth = chaikin(route, t=SMOOTHING_T, iterations=SMOOTHING_ITER)
    print(f"[smooth] Original: {len(route)} points → Smoothed: {len(route_smooth)} points")
    
    # 8. KPI 계산
    print("\n=== CALCULATING KPI ===")
    total_points = len(hotspots) + len(non_cover_random)
    collected_points = sum(1 for h in hotspots if h.visited) + sum(1 for r in non_cover_random if r.visited)
    total_mass = sum(h.mass for h in hotspots)
    collected_mass = sum(h.collected_mass for h in hotspots)
    
    kpi = {
        'total_operation_time_h': (total_travel_time + total_collection_time) / 3600.0,
        'distance_km': total_distance_m / 1000.0,
        'travel_time_min': total_travel_time / 60.0,
        'collection_time_min': total_collection_time / 60.0,
        'total_points': total_points,
        'collected_points': collected_points,
        'point_rate': collected_points / max(1, total_points),
        'total_mass': total_mass,
        'collected_mass': collected_mass,
        'mass_rate': collected_mass / max(1.0, total_mass),
        'collision_count': collision_count,
        'visited_hotspots': visited_count  # (명칭 유지)
    }
    
    print_run_summary(kpi)
    
    # 9. 결과 저장
    print("\n=== SAVING RESULTS ===")
    result_json = {
        'route': [{'lon': lon, 'lat': lat} for lon, lat in route],
        'route_smooth': [{'lon': lon, 'lat': lat} for lon, lat in route_smooth],
        'kpi': kpi,
        'hotspots': [
            {
                'id': h.id,
                'lon': h.lon,
                'lat': h.lat,
                'visited': h.visited,
                'mass': h.mass,
                'collected_mass': h.collected_mass
            }
            for h in hotspots
        ]
    }
    
    result_path = os.path.join(RESULT_DIR, "route_result.json")
    with open(result_path, 'w', encoding='utf-8') as f:
        json.dump(result_json, f, indent=2, ensure_ascii=False)
    print(f"[result] saved: {result_path}")
    
    # 10. 지도 생성
    if folium:
        print("\n=== GENERATING MAP ===")
        save_map_html(route_smooth, hotspots, non_cover_random, risk_pts)
    
    return result_json

def print_run_summary(kpi: dict):
    print("\n=== RUN SUMMARY ===")
    print(f"- total op time     : {kpi['total_operation_time_h']:.2f} h")
    print(f"- distance          : {kpi['distance_km']:.2f} km")
    print(f"- travel time       : {kpi['travel_time_min']:.1f} min")
    print(f"- collection time   : {kpi['collection_time_min']:.1f} min")
    print(f"- point rate (포인트): {kpi['point_rate']*100:.1f}% ({kpi['collected_points']}/{kpi['total_points']})")
    print(f"- mass  rate (질량)  : {kpi['mass_rate']*100:.1f}% ({kpi['collected_mass']:.0f}/{kpi['total_mass']:.0f})")
    print(f"- collisions        : {kpi['collision_count']}")
    print(f"- visited hotspots  : {kpi['visited_hotspots']}")

def save_map_html(route_smooth: List[Tuple[float,float]],
                  hotspots: List[Hotspot],
                  rand_pts: List[TrashPoint],
                  risk_pts: List[Tuple[float,float,float]]):
    if not folium:
        print("[map] SKIP: folium not installed")
        return

    clat = float(np.mean([lat for _, lat in route_smooth]))
    clon = float(np.mean([lon for lon, _ in route_smooth]))
    m = folium.Map(location=[clat, clon], zoom_start=10, tiles="cartodbpositron")

    # Route
    fg_route = folium.FeatureGroup(name="Route (Smooth)", show=True)
    folium.PolyLine([(lat, lon) for lon, lat in route_smooth], weight=4, color='#0066ff').add_to(fg_route)
    # Start (green)
    folium.CircleMarker(location=[route_smooth[0][1], route_smooth[0][0]], radius=8,
                        color='#00ff00', fill=True, tooltip="START (Northernmost)").add_to(fg_route)
    # End (red)
    folium.CircleMarker(location=[route_smooth[-1][1], route_smooth[-1][0]], radius=8,
                        color='#ff0000', fill=True, tooltip="END").add_to(fg_route)
    fg_route.add_to(m)

    # Hotspots
    fg_hs = folium.FeatureGroup(name="Hotspots", show=True)
    for h in hotspots:
        color = "#ff0000" if h.visited else "#ff7f00"
        folium.CircleMarker([h.lat, h.lon], radius=6, color=color, fill=True,
                            tooltip=f"{h.id} | w={h.weight:.1f} | mass={h.mass:.1f} | coll={h.collected_mass:.1f}").add_to(fg_hs)
        capture_R = max(CAPTURE_MIN_R_M, h.radius_m * CAPTURE_FACTOR)
        folium.Circle([h.lat, h.lon], radius=capture_R, color="#ffaaaa", fill=False, opacity=0.3).add_to(fg_hs)
    fg_hs.add_to(m)

    # Random Trash (보라색, 기본 숨김)
    fg_rand = folium.FeatureGroup(name="Random Trash", show=False)
    for t in rand_pts:
        color = "#8000ff" if not t.visited else "#b380ff"
        folium.CircleMarker([t.lat, t.lon], radius=3, color=color, fill=True, opacity=0.6).add_to(fg_rand)
    fg_rand.add_to(m)

    # Risk points (기본 숨김)
    if risk_pts:
        fg_risk = folium.FeatureGroup(name="Risk points", show=False)
        for (lon, lat, rv) in risk_pts:
            col = "#ff0000" if rv >= 0.4 else "#ffaa00" if rv >= 0.25 else "#00aa88"
            folium.CircleMarker([lat, lon], radius=2.5, color=col, fill=True,
                                tooltip=f"risk={rv:.2f}").add_to(fg_risk)
        fg_risk.add_to(m)

    folium.LayerControl(collapsed=False).add_to(m)
    out_html = os.path.join(RESULT_DIR, "route_map.html")
    m.save(out_html)
    print(f"[map] saved: {out_html}")
